## 图片分割

-
原型：

```python
ai.image_cut(img_base64)

```

-
描述：将图片按训练结果切割成多个目标物体。

-
必选参数：

  - img_base64：进行 BASE64转换后的图片。

-
返回：

  - result：返回一组切割的元素值，里面每个列表对应一个目标物体的元素值。数据类型：list。

-
示例

```python
ai.image_cut(img_base64=img_base64)

```

将图片按训练结果切割成多个目标物体。

更多示例，详见图像识别DEMO。

## 图片分类

-
原型：

```python
ai.feature_image_classify(img_base64s, lables, class_num, flag)

```

-
描述：将图片加入数据集进行训练。

-
必选参数：

  - img_base64s：一组进行 BASE64转换后的图片，数据类型：list。

  - lables：当前数据集的标记数字，比如第一个数据集设置为[0]。

  - class_num：数据集总数，比如一共 3 个数据集，设置为 3。

  - flag：第一个数据集的第一张图片设置为 0，最后一个数据集的最后一张图片标记为2，中间的图片设置为1。

- 返回：

  - True：训练成功。

  - False：训练失败。

-
示例

示例详见图像识别DEMO。

## 图像识别Demo

- Demo1：图像识别

本demo包含：

1）采集数据前的背景校准（可排除背景对图片识别的影响）；

2）拍照采集数据，并训练生成特征模型；

3）利用模型识别图片。

```python
from DobotEDU import *
import base64
import cv2
import time

def to_base64(file_name):  #转化为Base64格式
    with open(file_name, 'rb') as f:
        base64_data = base64.b64encode(f.read())
        return base64_data

def get_image(file_name, timeout, port, flip=False):
    pic = util.get_image(timeout, port, flip)
    cv2.imwrite(file_name, pic)
    base64_image = to_base64(file_name).decode("utf-8")
    return base64_image

#图像识别
#背景照
back_ground = get_image("D:/1.png", 5, 0)
#背景校准
ai.set_background(back_ground)
cv2.waitKey(0)
#建立数据集
size = 3  #数据集数量
data_base = {}
for j in range(size):
    a = input("请输入标签：")  #数据集名字
    print(a)
    data_base[j] = a
    # 0, 1, 2
    a_t = input("请输入数据集的图片数量：")
    a_i= int(a_t)
    for i in range(a_i):
        img = get_image("D:/1.png", timeout=5, port=0)
        cv2.waitKey(0)
        if i == 0 and j == 0:
            res = ai.feature_image_classify([img], [j], size, 0)
            print(res)
        elif i == a_i-1 and j == size - 1:
            haha = ai.feature_image_classify([img], [j], size, 2)
        else:
            ai.feature_image_classify([img], [j], size, 1)
#拍照获取待识别照片
while True:
  print("data_base", data_base)
  image = get_image("D:/1.png", timeout=5, port=0)
  cv2.waitKey(0)
  #调用模型识别
  res = ai.feature_image_group(image)
  result = data_base[res]
  print("图像识别的结果：", result)

```

- Demo2：保存识别后的文件

本demo用于说明数据集保存、删除的操作方法。包含：

1）如果数据已经存在，可选择是否删除重新采集数据或者保留原有数据直接训练（不用重新采集数据）；

2）数据的保存路径（可自定义修改）；

3）更新或保留数据后继续识别并输出结果。

```python
#新增功能：保存识别后的文件
#图像识别

from DobotEDU import *
import base64
import cv2
import time
import os

def to_base64(file_name):  #转化为Base64格式
    with open(file_name, 'rb') as f:
        base64_data = base64.b64encode(f.read())
        return base64_data

def get_image(file_name, timeout, port, flip=False):
    pic = util.get_image(timeout, port, flip)
    cv2.imwrite(file_name, pic)
    base64_image = to_base64(file_name).decode("utf-8")
    return base64_image

#删除文件
def del_file(path):
    ls = os.listdir(path)
    for i in ls:
        c_path = os.path.join(path, i)
        if os.path.isdir(c_path):
            os.rmdir(c_path)
        else:
            os.remove(c_path)
#图像识别
#判断是否需要背景校准
while True:
    is_cal = input("是否需要背景校准?(y/n)")
    if (is_cal == "y") or (is_cal == "Y"):
        #背景照
        back_ground = get_image("D:/1.png", 5, 0)
        #背景校准
        ai.set_background(back_ground)
        cv2.waitKey(0)
        break
    elif (is_cal == "n") or (is_cal == "N"):
        break
    else:
        print("请填入y or n.")

my_path = "D:\\temp\\userdir"  #照片存放文件夹路径在此修改
if not (os.path.exists(my_path)):
    os.mkdir(my_path)
else:
    is_tra = input("已存在目录,是否直接训练已有数据?(y or n)")
    while True:
        if (is_tra == "y") or (is_tra == "Y"):
            #直接训练
            files_dir = os.listdir(my_path)
            size = len(files_dir)
            data_base = {}
            print("将会对以下文件夹中的数据进行训练:", files_dir)
            for j in range(size):
                data_base[j] = files_dir[j]
                sub_dir = os.path.join(my_path, files_dir[j])
                if os.path.isdir(sub_dir):
                    sub_files = os.listdir(sub_dir)
                    a_i = len(sub_files)
                    for i in range(a_i):
                        img = to_base64(os.path.join(sub_dir, sub_files[i])).decode("utf-8")
                        if i == 0 and j == 0:
                            res = ai.feature_image_classify([img], [j], size, 0)
                        elif i == a_i - 1 and j == size - 1:
                            haha = ai.feature_image_classify([img], [j], size, 2)
                        else:
                            ai.feature_image_classify([img], [j], size, 1)
            break
        elif (is_tra == "n") or (is_tra == "N"):
            #生成图片数据后训练
            is_del = input("是否删除目录下的所有图片数据?(y/n)")
            if (is_del == "y") or (is_del == "Y"):
                del_file(my_path)
                print("删除成功")
            size = int(input("请输入数据集的数量:"))
            data_base = {}
            for j in range(size):
                a = input("请输入标签：")  #数据集名字
                temp_path = my_path + ("\\%s" % a)
                if not os.path.exists(temp_path):
                    os.makedirs(temp_path)
                data_base[j] = a
                # 0, 1, 2
                a_t = input("请输入数据集的图片数量：")
                a_i = int(a_t)
                for i in range(a_i):
                    img = get_image(temp_path + "\\%d.png" % i,
                                    timeout=5,
                                    port=0)
                    cv2.waitKey(0)
                    if i == 0 and j == 0:
                        res = ai.feature_image_classify([img], [j], size, 0)
                    elif i == a_i - 1 and j == size - 1:
                        haha = ai.feature_image_classify([img], [j], size, 2)
                    else:
                        ai.feature_image_classify([img], [j], size, 1)
            break
        else:
            print("请填入y or n.")

#拍照获取待识别照片
while True:
    print("data_base", data_base)
    image = get_image("D:/1.png", timeout=5, port=0)
    cv2.waitKey(0)
    #调用模型识别
    res = ai.feature_image_group(image)
    result = data_base[res]
    print("图像识别的结果：", result)

```

                #cv2.imshow("cut image",cut)
                #cv2.waitKey(0)
                base64_data = to_base64('D:/cut.jpg').decode("utf-8")
                cut_all.append(base64_data)
          if len(cut_all) == 1:
                res = ai.color_image_classify([cut_all[0]], [0], size, 2)
                print("color_image_classify: ", res)
          elif len(cut_all) == 0:
                print("没有想存储的照片")
          else:
                for m in range(len(cut_all)):
